DBT Patient Simulation Training Module
Product Specification Document
Overview
Purpose: An interactive voice-based training simulation that allows DBT therapists, group facilitators, and trainees to practice communication skills with a realistic simulated BPD patient. The simulation provides real-time feedback by escalating or de-escalating emotional intensity based on the user's communication approach.
Target Users:

DBT therapists in training
Group therapy facilitators
Clinical supervisors conducting role-play exercises
Experienced therapists wanting to sharpen skills

Learning Objectives:

Practice validation strategies (V1-V6) in real-time
Experience consequences of invalidating communication
Learn to balance acceptance and change
Practice de-escalation without reinforcing dysfunction
Recognize triggers common in BPD presentations

Real-Time Intensity Adjustment Logic
javascriptfunction processUserInput(userStatement, currentState) {
  let intensityDelta = 0;
  let classification = "neutral";
  let strategies = [];
  
  // Check for high invalidation (most damaging)
  for (const pattern of triggerPatterns.highInvalidation) {
    if (pattern.test(userStatement)) {
      intensityDelta += 2;
      classification = "invalidating";
      strategies.push(pattern.source);
    }
  }
  
  // Check for moderate invalidation
  for (const pattern of triggerPatterns.moderateInvalidation) {
    if (pattern.test(userStatement)) {
      intensityDelta += 1;
      classification = "invalidating";
      strategies.push(pattern.source);
    }
  }
  
  // Check for validation
  for (const pattern of triggerPatterns.validation) {
    if (pattern.test(userStatement)) {
      intensityDelta -= 1;
      classification = "validating";
      strategies.push(pattern.source);
    }
  }
  
  // Check for high validation
  for (const pattern of triggerPatterns.highValidation) {
    if (pattern.test(userStatement)) {
      intensityDelta -= 1.5;
      classification = "validating";
      strategies.push(pattern.source);
    }
  }
  
  // Check for dialectical statements
  for (const pattern of triggerPatterns.dialectical) {
    if (pattern.test(userStatement)) {
      intensityDelta -= 1;
      classification = "validating";
      strategies.push("dialectical");
    }
  }
  
  // Apply intensity change with bounds
  let newIntensity = Math.max(0, Math.min(9, currentState.currentIntensity + intensityDelta));
  
  // Track turns at level 9
  let turnsAtLevel9 = newIntensity >= 9 ? currentState.turnsAtLevel9 + 1 : 0;
  
  // Check exit conditions
  let outcome = null;
  if (newIntensity <= 0) {
    outcome = "success";
  } else if (turnsAtLevel9 >= 3) {
    outcome = "failure_offer_exit";
  }
  
  return {
    newIntensity: newIntensity,
    intensityDelta: intensityDelta,
    classification: classification,
    strategies: strategies,
    turnsAtLevel9: turnsAtLevel9,
    outcome: outcome
  };
}

javascriptfunction generateDebrief(simulationState) {
  const debrief = {
    outcome: simulationState.outcome,
    summary: "",
    effectiveStrategies: [],
    ineffectiveStrategies: [],
    alternatives: [],
    keyTakeaways: []
  };
  
  // Analyze history for patterns
  for (const turn of simulationState.history) {
    if (turn.classification === "validating") {
      debrief.effectiveStrategies.push({
        statement: turn.userStatement,
        strategy: identifyDBTStrategy(turn),
        effect: `Intensity decreased by ${Math.abs(turn.intensityChange)}`
      });
    } else if (turn.classification === "invalidating") {
      debrief.ineffectiveStrategies.push({
        statement: turn.userStatement,
        issue: identifyInvalidationType(turn),
        effect: `Intensity increased by ${turn.intensityChange}`,
        alternative: generateAlternative(turn)
      });
    }
  }
  
  // Generate key takeaways based on patterns
  if (debrief.ineffectiveStrategies.some(s => s.issue === "premature_problem_solving")) {
    debrief.keyTakeaways.push("Validate before problem-solving. The patient needs to feel heard before they can engage with solutions.");
  }
  
  if (debrief.ineffectiveStrategies.some(s => s.issue === "dismissing")) {
    debrief.keyTakeaways.push("Avoid minimizing language. Even well-intentioned attempts to help them 'feel better' can feel dismissive.");
  }
  
  // ... additional pattern analysis
  
  return debrief;
}

Exit Triggers
javascriptconst EXIT_PHRASE = "END DBT PATIENT SIMULATION";

function checkForSimulationExit(userInput, simulationState) {
  const normalized = userInput.toUpperCase().replace(/[^\w\s]/g, '');
  
  // Manual exit
  if (normalized.includes("END DBT PATIENT SIMULATION")) {
    return { exit: true, reason: "manual_exit" };
  }
  
  // Success exit
  if (simulationState.currentIntensity <= 0) {
    return { exit: true, reason: "success" };
  }
  
  // Failure exit offer (not forced)
  if (simulationState.turnsAtLevel9 >= 3) {
    return { exit: false, offerExit: true, reason: "sustained_crisis" };
  }
  
  return { exit: false };
}
